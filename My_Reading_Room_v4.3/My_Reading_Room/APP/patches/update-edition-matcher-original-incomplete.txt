const coverById = (id) => id ? 'https://covers.openlibrary.org/b/id/' + id + '-M.jpg' : '';
const coverByISBN = (isbn) => isbn ? 'https://covers.openlibrary.org/b/isbn/' + encodeURIComponent(isbn) + '-M.jpg?default=false' : '';
const digits = (str) => String(str || '').replace(/[^0-9X]/gi, '');
const cleanTitle = (t) => String(t || '').replace(/\s*\([^)]*#\d+[^)]*\)\s*$/, '').trim();
const normWords = (str) => new Set(String(str || '').toLowerCase().replace(/[^a-z0-9\s]/g, ' ').split(/\s+/).filter(w => w.length > 2 && !['the','and','for','with','from'].includes(w)));
const STOP_PUB = new Set(['books','book','press','publishing','publishers','publisher','pub','inc','llc','ltd','company','group','the','and','services','com','edition','editions']);
const pubTokens = (str) => [...new Set(String(str || '').toLowerCase().replace(/[^a-z0-9\s]/g, ' ').split(/\s+/).filter(w => w.length > 2 && !STOP_PUB.has(w)))];

export function normalizeBinding(str) {
  const s = String(str || '').toLowerCase();
  if (!s) return '';
  if (/kindle|ebook|e-book|digital|electronic/.test(s)) return 'ebook';
  if (/mass market/.test(s)) return 'mass_market';
  if (/paperback|softcover|trade paper/.test(s)) return 'paperback';
  if (/hardcover|hardback|cloth|leather/.test(s)) return 'hardcover';
  if (/audio/.test(s)) return 'audio';
  return s.trim();
}

export function scoreEditionMatch(target, candidate) {
  let score = 0;
  // 1. Publisher match (Publisher column)
  const tPub = pubTokens(target.publisher);
  const cPub = pubTokens(Array.isArray(candidate.publishers) ? candidate.publishers.join(' ') : (candidate.publisher || ''));
  if (tPub.length && cPub.length) {
    const overlap = tPub.filter(w => cPub.includes(w)).length;
    if (overlap > 0) score += 35 + Math.min(15, (overlap - 1) * 8);
  }
  // 2. Binding / physical format match (Binding column)
  const tBind = normalizeBinding(target.binding);
  const cBind = normalizeBinding(candidate.physical_format || candidate.binding || '');
  if (tBind && cBind) {
    if (tBind === cBind) score += 25;
    else if ((tBind === 'mass_market' && cBind === 'paperback') || (tBind === 'paperback' && cBind === 'mass_market')) score += 18;
  }
  // 3. Number of Pages proximity (Number of Pages column)
  const tPages = Number(target.page_count);
  const cPages = Number(candidate.page_count ?? candidate.number_of_pages);
  if (Number.isFinite(tPages) && tPages > 0 && Number.isFinite(cPages) && cPages > 0) {
    const diff = Math.abs(tPages - cPages);
    if (diff === 0) score += 40;
    else if (diff <= 5) score += 32;
    else if (diff <= 15) score += 22;
    else if (diff <= 35) score += 12;
    else if (diff <= 75) score += 5;
  }
  // 4. Year Published match (Year Published column vs edition publish_date)
  const tEdYear = Number(target.edition_year || target.published_year);
  const cYearStr = String(candidate.publish_date || candidate.published_year || '');
  const cYear = Number((cYearStr.match(/\d{4}/) || [])[0]);
  if (Number.isFinite(tEdYear) && tEdYear > 0 && Number.isFinite(cYear) && cYear > 0) {
    const yDiff = Math.abs(tEdYear - cYear);
    if (yDiff === 0) score += 30;
    else if (yDiff === 1) score += 20;
    else if (yDiff <= 3) score += 10;
  }
  // 5. Original Publication Year match (Original Publication Year column vs first_publish_year)
  const tOrigYear = Number(target.original_published_year || target.published_year);
  const cOrigYear = Number(candidate.first_publish_year || candidate.original_published_year || cYear);
  if (Number.isFinite(tOrigYear) && tOrigYear > 0 && Number.isFinite(cOrigYear) && cOrigYear > 0) {
    const oDiff = Math.abs(tOrigYear - cOrigYear);
    if (oDiff === 0) score += 20;
    else if (oDiff <= 2) score += 10;
  }
  if (candidate.cover_url || candidate.cover_i || (Array.isArray(candidate.covers) && candidate.covers[0])) score += 15;
  if (candidate.isbn || candidate.isbn_13?.[0] || candidate.isbn_10?.[0]) score += 8;
  return score;
}

function titleAuthorMatches(book, candidateTitle, candidateAuthors) {
  const bWords = normWords(cleanTitle(book.title));
  const cWords = normWords(cleanTitle(candidateTitle));
  if (!bWords.size || !cWords.size) return false;
  if (/\b(study guide|summary|analysis|workbook|companion|trivia)\b/i.test(candidateTitle) && !/\b(study guide|summary|analysis|workbook|companion|trivia)\b/i.test(book.title)) {
    return false;
  }
  let overlap = 0;
  for (const w of bWords) if (cWords.has(w)) overlap++;
  if (overlap / bWords.size < 0.5) return false;
  const aWords = normWords(book.author);
  const caWords = normWords(Array.isArray(candidateAuthors) ? candidateAuthors.join(' ') : candidateAuthors);
  if (aWords.size && caWords.size) {
    let aOverlap = 0;
    for (const w of aWords) if (caWords.has(w)) aOverlap++;
    if (aOverlap === 0) return false;
  }
  return true;
}

const GENRE_RULES = [
  [/science[-\s]?fiction|sci[-\s]?fi|space opera|cyberpunk|dystop/i, 'Science Fiction'],
  [/fantasy|sword and sorcery|epic fantasy|urban fantasy|magic/i, 'Fantasy'],
  [/thriller|suspense|espionage|spy/i, 'Thriller'],
  [/mystery|detective|whodunit|crime fiction/i, 'Mystery'],
  [/horror|ghost|vampire|zombie|supernatural/i, 'Horror'],
  [/historical fiction/i, 'Historical Fiction'],
  [/romance|love stor/i, 'Romance'],
  [/western/i, 'Western'],
  [/classic|literature/i, 'Classics'],
  [/adventure|survival/i, 'Adventure'],
  [/biograph|autobiograph/i, 'Biography'],
  [/\bmemoir\b/i, 'Memoir'],
  [/true crime/i, 'True Crime'],
  [/psycholog|mental health|cognitive|behavioral/i, 'Psychology'],
  [/self[-\s]?help|personal growth|habit|productivity|motivation|leadership/i, 'Self-Help'],
  [/business|economics|finance|investing|management|entrepreneur/i, 'Business & Finance'],
  [/history|world war|civil war|ancient|military/i, 'History'],
  [/sport|athlet|basketball|football|baseball|running|cycling|fitness|strength training|exercise/i, 'Sports & Fitness'],
  [/health|nutrition|medicine|medical|sleep|longevity|neuroscience/i, 'Health & Science'],
  [/\b(science|physics|biology|astronomy|chemistry|mathematics)\b/i, 'Science'],
  [/philosoph|stoic|ethics/i, 'Philosophy'],
  [/politic|government|sociology|social science/i, 'Politics & Society'],
  [/graphic novel|comic|manga/i, 'Graphic Novels'],
  [/poetry|poems/i, 'Poetry'],
  [/non[-\s]?fiction/i, 'Nonfiction'],
  [/^fiction$/i, 'Fiction']
];

export function normalizeGenres(rawList) {
  if (!Array.isArray(rawList)) return [];
  const found = [];
  const seen = new Set();
  for (const raw of rawList) {
    if (typeof raw !== 'string') continue;
    for (const part of raw.split(/[/,;]+/)) {
      const cleaned = part.trim();
      if (!cleaned || cleaned.length > 45) continue;
      for (const [pattern, label] of GENRE_RULES) {
        if (pattern.test(cleaned) && !seen.has(label)) {
          seen.add(label);
          found.push(label);
          break;
        }
      }
    }
  }
  if (found.length > 1 && found.includes('Fiction')) {
    return found.filter(g => g !== 'Fiction').slice(0, 4);
  }
  return found.slice(0, 4);
}

async function getJSON(url, timeout = 12000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeout);
  try {
    const response = await fetch(url, { signal: controller.signal });
    if (!response.ok) throw new Error('Catalog lookup failed (' + response.status + ')');
    return await response.json();
  } finally { clearTimeout(timer); }
}

function OLwork(doc) {
  return {
    kind: 'work', source: 'openlibrary', key: doc.key,
    title: doc.title || 'Untitled', author: (doc.author_name || []).join(', ') || 'Unknown author',
    published_year: doc.first_publish_year || null,
    cover_url: coverById(doc.cover_i), page_count: doc.number_of_pages_median || null,
    description: '', isbn: (doc.isbn || [])[0] || '',
    genres: normalizeGenres(doc.subject),
    edition_count: doc.edition_count || null
  };
}

function OLedition(doc, fallback = {}) {
  const isbn = doc.isbn_13?.[0] || doc.isbn_10?.[0] || fallback.isbn || '';
  const work = doc.works?.[0]?.key || fallback.work_key || '';
  const desc = typeof doc.description === 'string' ? doc.description : doc.description?.value;
  const pub = doc.publish_date || '';
  const year = (pub.match(/\d{4}/) || [])[0];
  const mappedGenres = normalizeGenres(doc.subjects);
  return {
    kind: 'edition', source: 'openlibrary', key: doc.key || '',
    edition_key: doc.key || '', work_key: work,
    title: doc.title || fallback.title || 'Untitled',
    author: fallback.author || 'Unknown author',
    published_year: year ? Number(year) : (fallback.published_year || null),
    page_count: doc.number_of_pages || null,
    publisher: (doc.publishers || []).join(', '),
    binding: doc.physical_format || fallback.binding || '',
    isbn, description: desc || fallback.description || '',
    cover_url: coverById(doc.covers?.[0]) || coverByISBN(isbn) || fallback.cover_url || '',
    genres: mappedGenres.length ? mappedGenres : (fallback.genres || []),
    edition_count: 1
  };
}

function googleVolume(item) {
  const v = item.volumeInfo || {};
  const ids = v.industryIdentifiers || [];
  const isbn = ids.find(i => i.type === 'ISBN_13')?.identifier || ids[0]?.identifier || '';
  return {
    kind:'edition', source:'googlebooks', key:item.id,
    work_key:'', edition_key:'google:' + item.id, isbn,
    title:v.title || 'Untitled', author:(v.authors || []).join(', ') || 'Unknown author',
    description:(v.description || '').replace(/<[^>]+>/g,''),
    published_year:Number((v.publishedDate || '').slice(0,4)) || null,
    page_count:v.pageCount || null, publisher:v.publisher || '',
    binding: item.saleInfo?.isEbook ? 'ebook' : (v.printType || ''),
    genres:normalizeGenres(v.categories),
    cover_url:(v.imageLinks?.thumbnail || v.imageLinks?.smallThumbnail || '').replace(/^http:/,'https:'),
    edition_count:1
  };
}

export async function searchCatalog(query, googleKey = '') {
  const q = query.trim();
  if (!q) return [];
  const isbn = digits(q);
  if (isbn === q.replace(/[-\s]/g, '') && (isbn.length === 10 || isbn.length === 13)) {
    try {
      const data = await getJSON(OPEN_LIBRARY + '/isbn/' + encodeURIComponent(isbn) + '.json');
      return [OLedition(data, { isbn })];
    } catch (_) { /* Use text search / Google fallback instead. */ }
  }
  try {
    const params = new URLSearchParams({ q, limit: '15', fields: 'key,title,author_name,first_publish_year,cover_i,isbn,subject,edition_count,number_of_pages_median' });
    const data = await getJSON(OPEN_LIBRARY + '/search.json?' + params);
    const list = (data.docs || []).filter(d => d.key && d.title).map(OLwork);
    if (list.length) return list;
  } catch (error) {
    if (!googleKey) throw error;
  }
  if (!googleKey) return [];
  const p = new URLSearchParams({ q: isbn.length === 10 || isbn.length === 13 ? 'isbn:' + isbn : q, maxResults:'15', key:googleKey });
  const result = await getJSON('https://www.googleapis.com/books/v1/volumes?' + p);
  return (result.items || []).map(googleVolume);
}

export async function getEditions(work) {
  if (work.kind === 'edition') return [work];
  const data = await getJSON(OPEN_LIBRARY + work.key + '/editions.json?limit=50');
  const editions = (data.entries || []).map(e => OLedition(e, {
    work_key: work.key, title: work.title, author: work.author,
    published_year: work.published_year, cover_url: work.cover_url,
    genres: work.genres || []
  }));
  return editions.sort((a,b) => (Number(Boolean(b.page_count)) - Number(Boolean(a.page_count))) || ((b.published_year || 0) - (a.published_year || 0)));
}

export async function enrichBookMetadata(book, googleKey = '', { preserveCover = true, forceGenres = false } = {}) {
  let isbn = digits(book.isbn);
  let cover_url = preserveCover && book.cover_url ? book.cover_url : '';
  let genres = !forceGenres && Array.isArray(book.genres) && book.genres.length ? book.genres : [];
  let work_key = book.work_key || null;
  let edition_key = book.edition_key || null;
  let description = book.description || null;
  let publisher = book.publisher || null;
  let page_count = book.page_count || null;
  let published_year = book.published_year || book.original_published_year || book.edition_year || null;
  const title = cleanTitle(book.title);

  // 1. Exact ISBN edition lookup on Open Library when ISBN is known
  if (isbn && (!cover_url || !genres.length || !description)) {
    try {
      const ed = await getJSON(OPEN_LIBRARY + '/isbn/' + encodeURIComponent(isbn) + '.json');
      if (!work_key && ed.works?.[0]?.key) work_key = ed.works[0].key;
      if (!edition_key && ed.key) edition_key = ed.key;
      if (!cover_url && ed.covers?.[0]) cover_url = coverById(ed.covers[0]);
      if (!genres.length && Array.isArray(ed.subjects)) genres = normalizeGenres(ed.subjects);
      if (!publisher && ed.publishers?.[0]) publisher = ed.publishers.join(', ');
      if (!page_count && ed.number_of_pages) page_count = ed.number_of_pages;
      const desc = typeof ed.description === 'string' ? ed.description : ed.description?.value;
      if (!description && desc) description = desc;
    } catch (_) { /* Fall back to search */ }
  }

  // 2. When ISBN is missing (or cover/genres/edition details still needed), find work on Open Library
  //    and score its editions using Publisher, Binding, Number of Pages, Year Published, and Original Publication Year.
  if ((!isbn || !cover_url || !genres.length) && title) {
    try {
      const params = new URLSearchParams({
        title,
        limit: '8',
        fields: 'key,title,author_name,first_publish_year,cover_i,subject,publisher,number_of_pages_median,isbn'
      });
      if (book.author) params.set('author', book.author);
      const data = await getJSON(OPEN_LIBRARY + '/search.json?' + params);
      const matches = (data.docs || []).filter(d => titleAuthorMatches(book, d.title, d.author_name));
      if (matches.length) {
        matches.sort((a, b) => scoreEditionMatch(book, {
          publishers: a.publisher,
          page_count: a.number_of_pages_median,
          first_publish_year: a.first_publish_year,
          cover_i: a.cover_i,
          isbn: a.isbn?.[0]
        }) - scoreEditionMatch(book, {
          publishers: b.publisher,
          page_count: b.number_of_pages_median,
          first_publish_year: b.first_publish_year,
          cover_i: b.cover_i,
          isbn: b.isbn?.[0]
        }));
        const bestWork = matches.at(-1);
        const genreHit = matches.find(d => normalizeGenres(d.subject).length);
        if (!work_key && bestWork.key) work_key = bestWork.key;
        if (!genres.length && genreHit) genres = normalizeGenres(genreHit.subject);
        if (!published_year && bestWork.first_publish_year) published_year = bestWork.first_publish_year;

        // Fetch work editions and rank by Publisher, Binding, Number of Pages, Year Published, and Original Year
        if (work_key && (!isbn || !cover_url || !edition_key)) {
          try {
            const edData = await getJSON(OPEN_LIBRARY + work_key + '/editions.json?limit=50');
            const entries = (edData.entries || []).map(e => ({
              raw: e,
              parsed: OLedition(e, {
                work_key,
                title: bestWork.title,
                author: book.author,
                published_year: bestWork.first_publish_year,
                cover_url: coverById(bestWork.cover_i),
                genres
              }),
              score: scoreEditionMatch(book, {
                ...e,
                first_publish_year: bestWork.first_publish_year
              })
            })).sort((a, b) => b.score - a.score);

            const bestEd = entries[0]?.parsed;
            const bestCoverEd = entries.find(x => x.raw.covers?.[0] || x.parsed.isbn)?.parsed;
            if (bestEd) {
              if (!edition_key && bestEd.edition_key) edition_key = bestEd.edition_key;
              if (!isbn && bestEd.isbn) isbn = digits(bestEd.isbn);
              if (!publisher && bestEd.publisher) publisher = bestEd.publisher;
              if (!page_count && bestEd.page_count) page_count = bestEd.page_count;
              if (!description && bestEd.description) description = bestEd.description;
              if (!genres.length && bestEd.genres?.length) genres = bestEd.genres;
            }
            if (!cover_url && bestCoverEd?.cover_url) cover_url = bestCoverEd.cover_url;
          } catch (_) { /* Ignore edition list fetch errors */ }
        }
        if (!cover_url) {
          const coverDoc = matches.find(d => d.cover_i) || bestWork;
          if (coverDoc?.cover_i) cover_url = coverById(coverDoc.cover_i);
        }
      }
    } catch (_) { /* Fall back to Google Books */ }
  }

  // 3. Google Books volume lookup scored by Publisher, Binding, Pages, Year Published, and Original Year
  if ((!cover_url || !genres.length || !isbn || !description) && (isbn || title)) {
    try {
      const queries = [];
      if (isbn) queries.push('isbn:' + isbn);
      if (title) {
        const pub = pubTokens(book.publisher)[0] || '';
        if (pub) queries.push('intitle:' + title + (book.author ? '+inauthor:' + book.author : '') + '+inpublisher:' + pub);
        queries.push('intitle:' + title + (book.author ? '+inauthor:' + book.author : ''));
      }
      for (const q of queries) {
        if (cover_url && genres.length && isbn && description) break;
        const p = new URLSearchParams({ q, maxResults: '8' });
        if (googleKey) p.set('key', googleKey);
        const res = await getJSON('https://www.googleapis.com/books/v1/volumes?' + p);
        const candidates = (res.items || [])
          .filter(it => isbn || titleAuthorMatches(book, it.volumeInfo?.title, it.volumeInfo?.authors))
          .map(it => {
            const vol = googleVolume(it);
            const score = scoreEditionMatch(book, {
              publisher: vol.publisher,
              binding: vol.binding,
              page_count: vol.page_count,
              publish_date: it.volumeInfo?.publishedDate,
              cover_url: vol.cover_url,
              isbn: vol.isbn
            });
            return { vol, score };
          })
          .sort((a, b) => b.score - a.score);

        for (const { vol } of candidates) {
          if (!cover_url && vol.cover_url) cover_url = vol.cover_url;
          if (!genres.length && vol.genres?.length) genres = vol.genres;
          if (!isbn && vol.isbn) isbn = digits(vol.isbn);
          if (!publisher &&
